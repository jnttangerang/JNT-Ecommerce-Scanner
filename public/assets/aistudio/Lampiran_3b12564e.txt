// Sesi per-karung (menggantikan sesi tanggal saja). Fallback baca key lama.
export type ReturSesi = { kodeBagging: string; dp: string; tglDatang: string; jumlahManifest: number };
const SESI_KEY = 'jt_retur_session';
export function getSesi(): ReturSesi | null {
  const raw = localStorage.getItem(SESI_KEY) ?? localStorage.getItem('jt_retur_session_tgldatang');
  if (!raw) return null;
  try { return JSON.parse(raw); }                 // format baru (objek)
  catch { return { kodeBagging:'', dp:'', tglDatang: raw, jumlahManifest: 0 }; } // key lama = string tanggal
}
export const setSesi = (s: ReturSesi) => localStorage.setItem(SESI_KEY, JSON.stringify(s));
